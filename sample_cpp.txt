#include <iostream>
using namespace std;

class Rectangle {
public:
    int width;
    int height;

    int area() {
        return width * height;
    }
};

int main() {
    int a = 4;
    int b = 6;

    if (a < b) {
        cout << "a is less than b" << endl;
    } else {
        cout << "a is not less than b" << endl;
    }

    for (int i = 0; i < 3; i++) {
        cout << "i = " << i << endl;
    }

    int x = 1;
    while (x <= 5) {
        cout << "x = " << x << endl;
        x++;
    }

    return 0;
}
